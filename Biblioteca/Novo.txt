// === IndexedDB ===
let db;
const request = indexedDB.open("RemissaoLeituraDB", 7);
request.onupgradeneeded = e => {
  db = e.target.result;
  if (!db.objectStoreNames.contains("recuperandos")) db.createObjectStore("recuperandos", {keyPath: "matricula"});
  if (!db.objectStoreNames.contains("livros")) db.createObjectStore("livros", {keyPath: "id", autoIncrement: true});
  if (!db.objectStoreNames.contains("leituras")) db.createObjectStore("leituras", {keyPath: "id", autoIncrement: true});
};
request.onsuccess = e => { db = e.target.result; carregarTudo(); };

// === Dados em memória ===
let recuperandos = [], livros = [], leituras = [];

// === Funções Auxiliares ===
function numeroPorExtenso(n) {
  const u = ["zero","um","dois","três","quatro","cinco","seis","sete","oito","nove","dez","onze","doze","treze","catorze","quinze","dezesseis","dezessete","dezoito","dezenove"];
  const d = ["","","vinte","trinta","quarenta","cinquenta"];
  if (n <= 19) return u[n];
  if (n <= 99) return d[Math.floor(n/10)] + (n%10>0?" e "+u[n%10]:"");
  return n+"";
}

// === Recuperandos ===
function salvarRecuperando() {
  const nome = document.getElementById("nomeRec").value.trim().toUpperCase();
  const matricula = document.getElementById("matriculaRec").value.trim();
  if (!nome || !matricula) return alert("Preencha nome e matrícula!");
  const rec = {
    matricula,
    nome,
    unidade: document.getElementById("unidadeRec").value,
    regime: document.getElementById("regimeRec").value
  };
  db.transaction("recuperandos","readwrite").objectStore("recuperandos").put(rec).onsuccess = () => {
    alert("Recuperando salvo!");
    document.getElementById("nomeRec").value = "";
    document.getElementById("matriculaRec").value = "";
    carregarTudo();
  };
}

// === Livros ===
function salvarLivro() {
  const titulo = document.getElementById("tituloLivro").value.trim();
  const autor = document.getElementById("autorLivro").value.trim();
  if (!titulo || !autor) return alert("Título e autor são obrigatórios!");
  const livro = {
    titulo, autor,
    isbn: document.getElementById("isbnLivro").value.trim() || "Sem ISBN",
    ano: parseInt(document.getElementById("anoLivro").value) || new Date().getFullYear(),
    categoria: document.getElementById("categoriaLivro").value.trim() || "Geral",
    quantidadeTotal: parseInt(document.getElementById("quantidadeLivro").value) || 1,
    quantidadeDisponivel: parseInt(document.getElementById("quantidadeLivro").value) || 1
  };
  db.transaction("livros","readwrite").objectStore("livros").add(livro).onsuccess = () => {
    alert("Livro adicionado!");
    limparFormularioLivro();
    carregarTudo();
  };
}
function limparFormularioLivro() {
  ["tituloLivro","autorLivro","isbnLivro","categoriaLivro"].forEach(id=>document.getElementById(id).value="");
  document.getElementById("anoLivro").value = new Date().getFullYear();
  document.getElementById("quantidadeLivro").value = "1";
}
function filtrarLivros() {
  const termo = document.getElementById("buscaLivro").value.toLowerCase();
  const filtrados = livros.filter(l => l.titulo.toLowerCase().includes(termo) || l.autor.toLowerCase().includes(termo) || (l.isbn && l.isbn.toLowerCase().includes(termo)));
  exibirLivros(filtrados);
}

// === Leitura ===
function salvarLeitura() {
  const matricula = document.getElementById("selectRecuperando").value;
  const livroId = document.getElementById("selectLivro").value;
  if (!matricula || !livroId) return alert("Selecione recuperando e livro!");
  const data = document.getElementById("dataLeitura").value;
  const ano = new Date(data).getFullYear();
  const leiturasAno = leituras.filter(l => l.matricula === matricula && new Date(l.dataLeitura).getFullYear() === ano && l.status === "aprovado");
  if (leiturasAno.length >= 12) return alert(`Limite de 12 livros aprovados em ${ano} atingido!`);
  const jaLeu = leituras.some(l => l.matricula === matricula && l.livroId === parseInt(livroId));
  if (jaLeu && !confirm("Este livro já foi registrado. Continuar mesmo assim?")) return;

  const livro = livros.find(l => l.id == livroId);
  const leitura = {
    matricula,
    nomeRecuperando: recuperandos.find(r=>r.matricula===matricula)?.nome || "Desconhecido",
    livroId: parseInt(livroId),
    titulo: livro.titulo,
    autor: livro.autor,
    dataLeitura: data,
    resenha: document.getElementById("resenha").value.trim(),
    status: document.getElementById("statusLeitura").value,
    diasRemidos: document.getElementById("statusLeitura").value === "aprovado" ? 4 : 0
  };
  db.transaction("leituras","readwrite").objectStore("leituras").add(leitura).onsuccess = () => {
    alert(leitura.status==="aprovado" ? "Aprovado! +4 dias remidos" : "Registrado como reprovado");
    document.getElementById("resenha").value = "";
    carregarTudo();
  };
}

// === Relatórios ===
function exportarRelatorioCompletoPDF() {
  const { jsPDF } = window.jspdf;
  const doc = new jsPDF('p','mm','a4');
  doc.setFont("helvetica","bold");
  doc.setFontSize(16);
  doc.text("RELATÓRIO DE REMISSÃO POR LEITURA - APAC",105,25,{align:"center"});
  doc.setFontSize(12);
  doc.text(`Gerado em: ${new Date().toLocaleString('pt-BR')}`,105,35,{align:"center"});
  doc.setFontSize(11);
  doc.text(`Recuperandos: ${recuperandos.length} • Livros: ${livros.length} • Aprovadas: ${leituras.filter(l=>l.status==="aprovado").length} • Total dias: ${leituras.filter(l=>l.status==="aprovado").length*4}`,20,50);

  const dados = recuperandos.map(r => {
    const qtd = leituras.filter(l=>l.matricula===r.matricula && l.status==="aprovado").length;
    return [r.nome, r.matricula, qtd, qtd*4];
  });
  doc.autoTable({head:[['Recuperando','Matrícula','Livros','Dias']], body:dados, startY:65, theme:'grid'});
  doc.text("_________________________________",60,doc.lastAutoTable.finalY + 30);
  doc.text("Assinatura do Responsável",60,doc.lastAutoTable.finalY + 37);
  doc.save(`Relatorio_APAC_${new Date().toISOString().slice(0,10)}.pdf`);
}

function gerarPDFIndividual(matricula) {
  const { jsPDF } = window.jspdf;
  const doc = new jsPDF();
  const rec = recuperandos.find(r=>r.matricula===matricula);
  const aprovadas = leituras.filter(l=>l.matricula===matricula && l.status==="aprovado")
    .sort((a,b)=>new Date(a.dataLeitura)-new Date(b.dataLeitura));

  doc.setFontSize(16);
  doc.text("RELATÓRIO INDIVIDUAL - REMISSÃO POR LEITURA",105,25,{align:"center"});
  doc.setFontSize(12);
  doc.text(`Nome: ${rec.nome}`,20,45);
  doc.text(`Matrícula: ${rec.matricula}`,20,52);
  doc.text(`Unidade: ${rec.unidade}`,20,59);
  doc.text(`Total de dias remidos: ${aprovadas.length*4}`,20,66);

  const tabela = aprovadas.map((l,i)=>[i+1, l.titulo, l.autor, new Date(l.dataLeitura).toLocaleDateString('pt-BR')]);
  doc.autoTable({head:[['Nº','Título','Autor','Data']], body:tabela, startY:80});
  doc.text("_________________________________",60,doc.lastAutoTable.finalY + 40);
  doc.text("Assinatura do Responsável",60,doc.lastAutoTable.finalY + 47);
  doc.save(`${rec.nome.split(' ')[0]}_${matricula}_remissao.pdf`);
}

  const tabela = aprovadas.map((l,i)=>[i+1, l.titulo.length>60?l.titulo.substr(0,57)+"..." : l.titulo, l.autor.length>30?l.autor.substr(0,27)+"..." : l.autor, new Date(l.dataLeitura).toLocaleDateString('pt-BR')]);
  doc.autoTable({head:[['Nº','Título da Obra','Autor','Conclusão']], body:tabela, startY:y, theme:'grid', headStyles:{fillColor:[139,0,139]}});

  y = doc.lastAutoTable.finalY + 25;
  doc.setFont("helvetica", "normal");
  doc.text("Nestes termos,", 20, y); y += 8;
  doc.text("Pede deferimento.", 20, y); y += 25;
  doc.text(configAPAC.cidade.split("/")[0] + ", " + new Date().toLocaleDateString('pt-BR') + ".", 20, y); y += 40;

  doc.line(30, y, 180, y);
  doc.text(configAPAC.presidente, 105, y + 8, {align:"center"});
  doc.text("Presidente da APAC", 105, y + 14, {align:"center"}); y += 40;

  doc.line(30, y, 180, y);
  doc.text(configAPAC.responsavel, 105, y + 8, {align:"center"});
  doc.text("Responsável pela Biblioteca", 105, y + 14, {align:"center"});

  doc.save(`OFICIO_${rec.nome.split(' ')[0].toUpperCase()}_${matricula}.pdf`);
}

// === Carregamento ===
function carregarTudo() {
  carregarRecuperandos(); carregarLivros(); carregarLeituras();
  document.getElementById("dataLeitura").valueAsDate = new Date();
}

function carregarRecuperandos() {
  db.transaction("recuperandos").objectStore("recuperandos").getAll().onsuccess = e => {
    recuperandos = e.target.result;
    document.getElementById("totalRec").textContent = recuperandos.length;
    const lista = document.getElementById("listaRecuperandos");
    const select = document.getElementById("selectRecuperando");
    lista.innerHTML = ""; select.innerHTML = "<option value=''>Selecione...</option>";
    recuperandos.forEach(r => {
      const lidas = leituras.filter(l=>l.matricula===r.matricula && l.status==="aprovado").length;
      lista.innerHTML += `
        <div style="padding:18px;border-bottom:1px solid #eee;background:${lidas>0?'#e8f5e9':'#fff'};position:relative;">
          <strong>${r.nome}</strong> • ${r.matricula}<br>
          <small>${lidas} livros • <strong>${lidas*4} dias remidos</strong></small>
          <button class="small danger" style="position:absolute;top:10px;right:10px;" onclick="confirmarExclusaoRecuperando('${r.matricula}', '${r.nome}')">Excluir</button>
        </div>`;
      select.innerHTML += `<option value="${r.matricula}">${r.nome} (${r.matricula})</option>`;
    });
  };
}

function carregarLivros() {
  db.transaction("livros").objectStore("livros").getAll().onsuccess = e => {
    livros = e.target.result;
    document.getElementById("totalLivrosAcervo").textContent = livros.length;
    exibirLivros(livros);
    atualizarSelectLivros();
  };
}

function exibirLivros(lista = livros) {
  const div = document.getElementById("listaLivros");
  div.innerHTML = lista.length===0 ? "<p style='text-align:center;color:#777;'>Nenhum livro cadastrado.</p>" : "";
  lista.forEach(livro => {
    const badge = livro.quantidadeDisponivel>0 ? `<span class="badge disponivel">Disponível (${livro.quantidadeDisponivel})</span>` : `<span class="badge indisponivel">Indisponível</span>`;
    div.innerHTML += `<div style="padding:18px;border:1px solid #eee;border-radius:8px;margin-bottom:12px;background:#fafafa;">
      <strong>${livro.titulo}</strong> - ${livro.autor}<br>
      <small>ISBN: ${livro.isbn!=="Sem ISBN"?livro.isbn:"Não informado"} | ${livro.ano} | ${livro.categoria}</small><br>${badge}
      <div style="margin-top:12px;">
        <button class="small danger" onclick="if(confirm('Excluir permanentemente?')) db.transaction('livros','readwrite').objectStore('livros').delete(${livro.id}).onsuccess=carregarTudo()">Excluir</button>
      </div>
    </div>`;
  });
}

function atualizarSelectLivros() {
  const select = document.getElementById("selectLivro");
  select.innerHTML = "<option value=''>Selecione...</option>";
  livros.filter(l=>l.quantidadeDisponivel>0).forEach(l => {
    select.innerHTML += `<option value="${l.id}">${l.titulo} - ${l.autor}</option>`;
  });
}

function carregarLeituras() {
  db.transaction("leituras").objectStore("leituras").getAll().onsuccess = e => {
    leituras = e.target.result.sort((a,b)=>new Date(b.dataLeitura)-new Date(a.dataLeitura));
    const tabela = document.getElementById("listaLeituras");
    tabela.innerHTML = `<table><tr><th>Data</th><th>Recuperando</th><th>Livro</th><th>Status</th><th>Dias</th></tr></table>`;
    const tbody = document.createElement("tbody");
    leituras.forEach(l => {
      const row = tbody.insertRow();
      row.style.background = l.status==="aprovado" ? "#d4edda" : "#f8d7da";
      row.innerHTML = `<td>${new Date(l.dataLeitura).toLocaleDateString('pt-BR')}</td><td>${l.nomeRecuperando}</td><td>${l.titulo}</td><td>${l.status==="aprovado"?"Aprovado":"Reprovado"}</td><td>${l.diasRemidos}</td>`;
    });
    tabela.appendChild(tbody);
    atualizarRelatorios();
  };
}

function atualizarRelatorios() {
  const aprovadas = leituras.filter(l=>l.status==="aprovado");
  document.getElementById("totalRecuperandos").textContent = recuperandos.length;
  document.getElementById("totalLivrosAcervo").textContent = livros.length;
  document.getElementById("totalLeiturasAprovadas").textContent = aprovadas.length;
  document.getElementById("totalDiasRemidos").textContent = aprovadas.length*4;

  const resumo = document.getElementById("resumoPorPessoa");
  resumo.innerHTML = `<table><tr><th>PDF</th><th>Recuperando</th><th>Livros</th><th>Dias</th></tr>`;
  recuperandos.forEach(r => {
    const qtd = leituras.filter(l=>l.matricula===r.matricula && l.status==="aprovado").length;
    resumo.innerHTML += `<tr>
      <td><button class="small pdf" onclick="gerarPDFIndividual('${r.matricula}')">PDF</button></td>
      <td><strong>${r.nome}</strong><br><small>${r.matricula}</small></td>
      <td>${qtd}</td>
      <td><strong>${qtd*4}</strong></td>
    </tr>`;
  });
  resumo.innerHTML += "</table>";
}

document.getElementById("fileInput").onchange = e => {
  const file = e.target.files[0];
  if (!file) return;
  const reader = new FileReader();
  reader.onload = ev => {
    try {
      const dados = JSON.parse(ev.target.result);
      const tx = db.transaction(["recuperandos","livros","leituras"],"readwrite");
      ["recuperandos","livros","leituras"].forEach(store => tx.objectStore(store).clear());
      (dados.recuperandos||[]).forEach(r=>tx.objectStore("recuperandos").add(r));
      (dados.livros||[]).forEach(l=>tx.objectStore("livros").add(l));
      (dados.leituras||[]).forEach(l=>tx.objectStore("leituras").add(l));
      tx.oncomplete = () => { alert("Backup restaurado!"); carregarTudo(); };
    } catch(err) { alert("Erro: "+err.message); }
  };
  reader.readAsText(file);
};

// === Exclusão segura de recuperando ===
function confirmarExclusaoRecuperando(matricula, nome) {
  if (!confirm(`Excluir permanentemente:\n${nome}\nMatrícula: ${matricula}\n\nEsta ação não pode ser desfeita!`)) return;
  const senha = prompt("Digite EXCLUIR para confirmar:");
  if (senha !== "EXCLUIR") return alert("Cancelado.");
  const tx = db.transaction(["recuperandos","leituras"],"readwrite");
  tx.objectStore("recuperandos").delete(matricula);
  const leiturasStore = tx.objectStore("leituras");
  leiturasStore.getAll().onsuccess = e => {
    e.target.result.filter(l=>l.matricula===matricula).forEach(l=>leiturasStore.delete(l.id));
  };
  tx.oncomplete = () => { alert("Excluído com sucesso!"); carregarTudo(); };
}

function openTab(evt, tabName) {
  document.querySelectorAll(".tabcontent").forEach(t=>t.style.display="none");
  document.querySelectorAll(".tablink").forEach(t=>t.classList.remove("active"));
  document.getElementById(tabName).style.display="block";
  evt.currentTarget.classList.add("active");
  if (tabName==="leitura") atualizarSelectLivros();
}
